import json
entries = json.load(open("run.json"))

def is_post(e):
    h = e.get("httpRequest") or {}
    return h.get("requestMethod") == "POST"

turns, cur = [], None
for e in entries:
    if is_post(e):
        if cur: turns.append(cur)
        cur = {"post": e, "logs": []}
    elif cur is not None:
        cur["logs"].append(e)
if cur: turns.append(cur)
