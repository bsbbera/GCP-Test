from google.oauth2.credentials import Credentials
from google.cloud import datacatalog_lineage_v1

def get_upstream_lineage_with_token(project_id, location, target_table_fqn, access_token):
    """
    Fetches upstream tables using a raw OAuth Access Token.
    """
    # 1. Wrap the raw token string in a Credentials object
    creds = Credentials(token=access_token)
    
    # 2. Initialize the client, explicitly passing the credentials
    client = datacatalog_lineage_v1.LineageClient(credentials=creds)
    
    parent = f"projects/{project_id}/locations/{location}"
    target_entity = datacatalog_lineage_v1.EntityReference(
        fully_qualified_name=target_table_fqn
    )
    
    request = datacatalog_lineage_v1.SearchLinksRequest(
        parent=parent,
        target=target_entity
    )
    
    page_result = client.search_links(request=request)
    
    upstream_tables = []
    print(f"Upstream lineage for: {target_table_fqn}\n" + "-"*50)
    
    for link in page_result:
        source_fqn = link.source.fully_qualified_name
        upstream_tables.append(source_fqn)
        print(f"Source Table: {source_fqn}")
        
    return upstream_tables

# --- Usage ---
PROJECT_ID = "your-gcp-project-id"
LOCATION = "us" # e.g., 'us' or 'eu'
TARGET_FQN = "bigquery:your-gcp-project-id.your_dataset.your_table"

# Paste your BigQuery Auth Access Token string here

immediate_sources = get_upstream_lineage_with_token(
    project_id=PROJECT_ID, 
    location=LOCATION, 
    target_table_fqn=TARGET_FQN,

)
