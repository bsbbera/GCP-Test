CREATE SCHEMA IF NOT EXISTS dq;

CREATE OR REPLACE TABLE dq.table_metadata AS
WITH
tbl AS (
  SELECT CONCAT(table_catalog,'.',table_schema,'.',table_name) AS table_id,
         table_catalog AS project_name, table_schema AS dataset_name, table_name,
         table_type, creation_time
  FROM `prd-amer-analyt-datal-svc-88.region-us`.INFORMATION_SCHEMA.TABLES
  WHERE REGEXP_CONTAINS(table_schema, r'_dc_|_d_|_h_')
),
cols AS (
  SELECT CONCAT(table_catalog,'.',table_schema,'.',table_name) AS table_id,
         COUNT(*) AS column_count,
         STRING_AGG(CONCAT(column_name,' (',data_type,')'), ', ' ORDER BY ordinal_position) AS column_list,
         STRING_AGG(IF(is_partitioning_column = 'YES', column_name, NULL)) AS partition_column
  FROM `prd-amer-analyt-datal-svc-88.region-us`.INFORMATION_SCHEMA.COLUMNS
  GROUP BY 1
),
ds_opt AS (
  SELECT CONCAT(catalog_name,'.',schema_name) AS ds_id,
         MAX(IF(option_name = 'description', TRIM(option_value,'"'), NULL)) AS dataset_description,
         MAX(IF(option_name = 'labels', option_value, NULL)) AS dataset_labels
  FROM `prd-amer-analyt-datal-svc-88.region-us`.INFORMATION_SCHEMA.SCHEMATA_OPTIONS
  GROUP BY 1
),
tbl_opt AS (
  SELECT CONCAT(table_catalog,'.',table_schema,'.',table_name) AS table_id,
         MAX(IF(option_name = 'description', TRIM(option_value,'"'), NULL)) AS table_description,
         MAX(IF(option_name = 'labels', option_value, NULL)) AS table_labels
  FROM `prd-amer-analyt-datal-svc-88.region-us`.INFORMATION_SCHEMA.TABLE_OPTIONS
  GROUP BY 1
),
stor AS (
  SELECT CONCAT(project_id,'.',table_schema,'.',table_name) AS table_id,
         total_rows, total_partitions, total_logical_bytes, storage_last_modified_time
  FROM `prd-amer-analyt-datal-svc-88.region-us`.INFORMATION_SCHEMA.TABLE_STORAGE
  WHERE deleted = FALSE
),
jobs AS (
  SELECT *
  FROM `prd-amer-analyt-datal-svc-88.region-us`.INFORMATION_SCHEMA.JOBS
  WHERE creation_time >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 180 DAY)
),
reads AS (
  SELECT CONCAT(rt.project_id,'.',rt.dataset_id,'.',rt.table_id) AS table_id,
         COUNT(DISTINCT job_id) AS query_count_180d,
         MAX(creation_time) AS last_queried,
         COUNT(DISTINCT user_email) AS distinct_users,
         COUNTIF(error_result IS NOT NULL) AS failed_queries
  FROM jobs, UNNEST(referenced_tables) AS rt
  WHERE job_type = 'QUERY'
  GROUP BY 1
),
writes AS (
  SELECT CONCAT(destination_table.project_id,'.',destination_table.dataset_id,'.',destination_table.table_id) AS table_id,
         COUNTIF(error_result IS NULL) AS successful_loads_180d,
         COUNTIF(error_result IS NOT NULL) AS failed_loads_180d,
         MAX(IF(error_result IS NULL, creation_time, NULL)) AS last_load,
         SAFE_DIVIDE(
           TIMESTAMP_DIFF(MAX(IF(error_result IS NULL, creation_time, NULL)),
                          MIN(IF(error_result IS NULL, creation_time, NULL)), HOUR),
           COUNTIF(error_result IS NULL) - 1) AS avg_hrs,
         ARRAY_TO_STRING(ARRAY_AGG(DISTINCT user_email IGNORE NULLS LIMIT 3), ', ') AS writers
  FROM jobs
  WHERE destination_table.table_id IS NOT NULL
    AND (job_type IN ('LOAD','COPY')
         OR statement_type IN ('INSERT','MERGE','UPDATE','DELETE','CREATE_TABLE_AS_SELECT'))
  GROUP BY 1
),
edges AS (
  SELECT DISTINCT
    CONCAT(rt.project_id,'.',rt.dataset_id,'.',rt.table_id) AS src,
    CONCAT(destination_table.project_id,'.',destination_table.dataset_id,'.',destination_table.table_id) AS dst
  FROM jobs, UNNEST(referenced_tables) AS rt
  WHERE destination_table.table_id IS NOT NULL
    AND statement_type IN ('INSERT','MERGE','UPDATE','CREATE_TABLE_AS_SELECT')
    AND error_result IS NULL
),
upstream AS (
  SELECT dst AS table_id, STRING_AGG(src, ', ') AS upstream_tables
  FROM edges WHERE src != dst GROUP BY 1
),
downstream AS (
  SELECT src AS table_id, STRING_AGG(dst, ', ') AS downstream_tables
  FROM edges WHERE src != dst GROUP BY 1
)
SELECT
  t.project_name, t.dataset_name, t.table_name, t.table_type,
  d.dataset_description, o.table_description,
  c.column_count, c.column_list, c.partition_column,
  t.creation_time AS created_date,
  s.storage_last_modified_time AS last_updated,
  DATE_DIFF(CURRENT_DATE(), DATE(s.storage_last_modified_time), DAY) AS days_since_update,
  s.total_rows AS row_count, s.total_partitions,
  ROUND(s.total_logical_bytes / POW(1024,3), 3) AS total_gb,
  r.query_count_180d, r.last_queried, r.distinct_users, r.failed_queries,
  w.successful_loads_180d, w.failed_loads_180d, w.last_load,
  ROUND(w.avg_hrs, 1) AS avg_hrs_between_loads,
  CASE WHEN w.avg_hrs IS NULL THEN 'Unknown'
       WHEN w.avg_hrs <= 1.5 THEN 'Hourly'
       WHEN w.avg_hrs <= 30  THEN 'Daily'
       WHEN w.avg_hrs <= 200 THEN 'Weekly'
       ELSE 'Monthly+' END AS load_frequency,
  u.upstream_tables, dn.downstream_tables,
  COALESCE(REGEXP_EXTRACT(o.table_labels,  r'"owner", "([^"]+)"'),
           REGEXP_EXTRACT(d.dataset_labels, r'"owner", "([^"]+)"')) AS owner_label,
  w.writers, o.table_labels,
  r.last_queried IS NULL AS is_unused_180d,
  DATE_DIFF(CURRENT_DATE(), DATE(s.storage_last_modified_time), DAY) > 30 AS is_stale_30d,
  o.table_description IS NOT NULL AS is_documented
FROM tbl t
LEFT JOIN cols c        USING (table_id)
LEFT JOIN ds_opt d      ON d.ds_id = CONCAT(t.project_name,'.',t.dataset_name)
LEFT JOIN tbl_opt o     USING (table_id)
LEFT JOIN stor s        USING (table_id)
LEFT JOIN reads r       USING (table_id)
LEFT JOIN writes w      USING (table_id)
LEFT JOIN upstream u    USING (table_id)




CREATE OR REPLACE TABLE dq.table_metadata_clean AS
SELECT
  *,
  REGEXP_CONTAINS(
    table_name,
    r'(?i)(?:^|_)(?:'
    r'[a-z0-9]*(?:backup|bkp|bak|temp|tmp|demo|test|mock|before|after|original|missing|miss|lastmodify|fullload|null|befores4|copy|sp_status|error)[a-z0-9]*'
    r'|(?:exp|cte|old|copy|dup|bkup|chg)\d*'
    r'|[a-z]*\d{4,}[a-z0-9]*'
    r'|\d{1,2}(?:st|nd|rd|th)?_?(?:jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)[a-z]*(?:_?\d{2,4})?'
    r'|(?:jan|feb|mar|apr|may|jun|jul|aug|sep|oct|nov|dec)[a-z]*_?\d{1,4}'
    r')(?:_|$)'
  ) AS is_junk
FROM dq.table_metadata
WHERE table_type = 'BASE TABLE'
  AND NOT STARTS_WITH(table_name, 'v_')
  AND TRIM(IFNULL(table_name, '')) != '';

-- Kept (data_proj_filt4)
SELECT * FROM dq.table_metadata_clean WHERE NOT is_junk;

-- Removed (removed)
SELECT * FROM dq.table_metadata_clean WHERE is_junk;
LEFT JOIN downstream dn USING (table_id);
