#!/usr/bin/env python3
"""
Cloud Run chatbot eval extract — items 3, 4, 7, 8 only.

    pip install google-cloud-logging
    python3 eval_runs.py --from "17:35"
    python3 eval_runs.py --last 60
    python3 eval_runs.py --file run.json

Outputs:
    turns.csv    one row per prompt  (model, times, tools, tokens, bytes, latency split)
    stages.csv   one row per stage   (item 8, full detail)
"""

import json
import sys
import csv
import re
import argparse
from datetime import datetime, timedelta, timezone

PROJECT = "ul-da-d-931978-prj"
SERVICE = "search-copilot-agent"
IST = timezone(timedelta(hours=5, minutes=30))

BYTE_HINTS = ("bytes_processed", "bytes_billed", "bytes_scanned", "bytesprocessed")


# ---------------------------------------------------------------- fetch
def fetch(start, end, limit=5000):
    from google.cloud import logging_v2
    flt = (f'resource.type="cloud_run_revision" '
           f'resource.labels.service_name="{SERVICE}" '
           f'timestamp>="{start.astimezone(timezone.utc).isoformat()}"')
    if end:
        flt += f' timestamp<="{end.astimezone(timezone.utc).isoformat()}"'
    client = logging_v2.Client(project=PROJECT)
    out = []
    for i, e in enumerate(client.list_entries(filter_=flt,
                                              order_by=logging_v2.ASCENDING,
                                              page_size=1000)):
        out.append(e.to_api_repr())
        if i + 1 >= limit:
            break
    print(f"[{len(out)} log entries fetched]")
    return out


# ---------------------------------------------------------------- helpers
def ts(e):
    return datetime.fromisoformat(e["timestamp"].replace("Z", "+00:00"))


def ev_and_extra(e):
    """Return (event_text, extra_dict). Handles JSON and plain-text lines."""
    p = e.get("jsonPayload")
    if isinstance(p, dict):
        extra = p.get("extra")
        return str(p.get("event") or p.get("message") or ""), \
            (extra if isinstance(extra, dict) else {})
    return str(e.get("textPayload") or ""), {}


def num(v):
    try:
        return float(str(v).replace(",", ""))
    except (TypeError, ValueError):
        return None


def find_num(extra, must_have, exclude=()):
    """First numeric value whose key contains all of must_have."""
    for k, v in extra.items():
        kl = str(k).lower()
        if all(w in kl for w in must_have) and not any(x in kl for x in exclude):
            n = num(v)
            if n is not None:
                return n
    return None


# ---------------------------------------------------------------- split
def split_turns(entries):
    entries = sorted(entries, key=ts)
    turns, cur = [], None
    for e in entries:
        h = e.get("httpRequest") or {}
        if h.get("requestMethod") == "POST":
            lat = float(str(h.get("latency", "0s")).rstrip("s"))
            cur = {"start": ts(e), "end": ts(e) + timedelta(seconds=lat),
                   "total_s": round(lat, 2), "status": h.get("status"), "logs": []}
            turns.append(cur)
        elif not h and cur is not None:
            cur["logs"].append(e)
    if not turns:
        sys.exit("No POST entries found — widen the time window.")
    return turns


# ---------------------------------------------------------------- extract
def build_stages(turn, idx):
    """Item 8: named stages with real durations."""
    marks = [(ts(e), *ev_and_extra(e)) for e in turn["logs"]]
    stages = []

    # LLM: 'Sending out request' -> next 'ADK LLM call timing'
    pending = None
    for t, ev, _ in marks:
        if "Sending out request" in ev:
            pending = t
        elif "ADK LLM call timing" in ev and pending:
            stages.append(("llm_call", pending, t))
            pending = None

    # BigQuery: group of concurrent queries -> wall clock of the batch
    bq_open, bq_last = None, None
    for t, ev, _ in marks:
        if "Executing BigQuery query" in ev:
            if bq_open is None:
                bq_open = t
        elif "Query executed successfully" in ev and bq_open:
            bq_last = t
        elif "Sending out request" in ev and bq_open and bq_last:
            stages.append(("bq_batch", bq_open, bq_last))
            bq_open = bq_last = None
    if bq_open and bq_last:
        stages.append(("bq_batch", bq_open, bq_last))

    # front and back of the request
    if marks:
        stages.append(("api_startup", turn["start"], marks[0][0]))
        stages.append(("response_return", marks[-1][0], turn["end"]))

    stages.sort(key=lambda s: s[1])
    return [{"turn": idx, "stage": f"{n}_{i}" if n in ("llm_call", "bq_batch") else n,
             "start": a.astimezone(IST).strftime("%H:%M:%S.%f")[:-3],
             "end": b.astimezone(IST).strftime("%H:%M:%S.%f")[:-3],
             "duration_s": round((b - a).total_seconds(), 3)}
            for i, (n, a, b) in enumerate(stages, 1)]


def summarize(turn, idx):
    tools, model = [], ""
    prompt_tok = out_tok = total_tok = 0
    llm_calls = 0
    bq_queries = 0
    bq_bytes = 0
    bytes_seen = False

    for e in turn["logs"]:
        ev, extra = ev_and_extra(e)

        # item 3 — which tool was picked
        if ev.startswith("KPI tool invoked"):
            tools.append(ev.split(":", 1)[1].strip())

        # model name
        m = re.search(r"model:\s*([\w.\-]+)", ev)
        if m and not model:
            model = m.group(1)

        # item 4 — tokens and call count
        if "ADK LLM call timing" in ev:
            llm_calls += 1
            p = find_num(extra, ["prompt", "token"]) or \
                find_num(extra, ["input", "token"])
            o = find_num(extra, ["candidates", "token"]) or \
                find_num(extra, ["output", "token"]) or \
                find_num(extra, ["completion", "token"])
            t = find_num(extra, ["total", "token"])
            prompt_tok += p or 0
            out_tok += o or 0
            total_tok += t or 0

        # item 7 — bytes scanned, only if the app actually logs it
        if "BigQuery" in ev or "Query executed" in ev:
            if "Executing BigQuery query" in ev:
                bq_queries += 1
            for k, v in extra.items():
                if any(h in str(k).lower() for h in BYTE_HINTS):
                    n = num(v)
                    if n is not None:
                        bq_bytes += n
                        bytes_seen = True

    if not total_tok:
        total_tok = prompt_tok + out_tok

    return {
        "turn": idx,
        "start_time": turn["start"].astimezone(IST).strftime("%Y-%m-%d %H:%M:%S"),
        "end_time": turn["end"].astimezone(IST).strftime("%Y-%m-%d %H:%M:%S"),
        "duration_s": turn["total_s"],
        "status": turn["status"],
        "model": model,
        "n_tool_calls": len(tools),
        "tools_used": " | ".join(tools),
        "llm_call_count": llm_calls,
        "prompt_tokens": int(prompt_tok),
        "output_tokens": int(out_tok),
        "total_tokens": int(total_tok),
        "n_bq_queries": bq_queries,
        "bq_bytes_scanned": int(bq_bytes) if bytes_seen else "",
        "_bytes_seen": bytes_seen,
    }


# ---------------------------------------------------------------- output
def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--from", dest="frm", help="IST start, e.g. 17:35")
    ap.add_argument("--to", dest="to", help="IST end, e.g. 18:30")
    ap.add_argument("--last", type=int, help="last N minutes")
    ap.add_argument("--file", help="parse a saved JSON file instead")
    a = ap.parse_args()

    if a.file:
        entries = json.load(open(a.file))
    else:
        def at(hhmm):
            d = datetime.now(IST).date()
            h, m = (int(x) for x in hhmm.split(":"))
            return datetime(d.year, d.month, d.day, h, m, tzinfo=IST)
        if a.last:
            start, end = datetime.now(IST) - timedelta(minutes=a.last), None
        elif a.frm:
            start, end = at(a.frm), at(a.to) if a.to else None
        else:
            start, end = datetime.now(IST) - timedelta(minutes=60), None
        entries = fetch(start, end)

    raw = split_turns(entries)
    rows = [summarize(t, i) for i, t in enumerate(raw, 1)]
    stages = [s for i, t in enumerate(raw, 1) for s in build_stages(t, i)]

    keep_bytes = any(r.pop("_bytes_seen") for r in rows)
    if not keep_bytes:
        for r in rows:
            r.pop("bq_bytes_scanned")
        print("[bq_bytes_scanned not present in logs — column dropped]")

    for r in rows:
        print("-" * 66)
        print(f"TURN {r['turn']}  {r['start_time']} -> {r['end_time'][-8:]}  "
              f"{r['duration_s']}s  HTTP {r['status']}  {r['model']}")
        print(f"  tools     : {r['tools_used'] or '-'}")
        print(f"  llm       : {r['llm_call_count']} calls, "
              f"{r['prompt_tokens']} in / {r['output_tokens']} out "
              f"/ {r['total_tokens']} total")
        line = f"  bigquery  : {r['n_bq_queries']} queries"
        if keep_bytes:
            line += f", {r['bq_bytes_scanned']} bytes"
        print(line)
        print("  stages    :")
        for s in [s for s in stages if s["turn"] == r["turn"]]:
            print(f"     {s['stage']:18} {s['duration_s']:8.3f}s  "
                  f"{s['start']} -> {s['end']}")

    with open("turns.csv", "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=list(rows[0]))
        w.writeheader()
        w.writerows(rows)

    with open("stages.csv", "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=["turn", "stage", "start", "end",
                                          "duration_s"])
        w.writeheader()
        w.writerows(stages)

    print(f"\nturns.csv ({len(rows)} turns) | stages.csv ({len(stages)} stages)")


if __name__ == "__main__":
    main()
