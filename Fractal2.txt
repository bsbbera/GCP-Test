DECLARE region STRING DEFAULT 'region-us';
DECLARE ds     STRING DEFAULT NULL;        -- NULL = whole project


DECLARE proj   STRING DEFAULT 'my-proj';
DECLARE region STRING DEFAULT 'region-us';
DECLARE ds     STRING DEFAULT 'sales';     -- your dataset


DECLARE isq   STRING DEFAULT FORMAT('`%s.%s`.INFORMATION_SCHEMA', proj, region);
DECLARE scope STRING DEFAULT CONCAT(proj, '.', IF(ds IS NULL, '', ds || '.'));

-- 1. SETUP
CREATE SCHEMA IF NOT EXISTS dq;
CREATE TABLE IF NOT EXISTS dq.table_profile (
  table_id STRING, total_rows INT64, dup_rows INT64, total_nulls INT64,
  null_by_column STRING, error STRING, run_ts TIMESTAMP);
DELETE FROM dq.table_profile WHERE STARTS_WITH(table_id, scope);

-- 2. TABLES IN SCOPE (columns picked automatically)
EXECUTE IMMEDIATE FORMAT("""
  CREATE OR REPLACE TABLE dq._work AS
  SELECT c.table_catalog p, c.table_schema d, c.table_name t,
    STRING_AGG(FORMAT('COUNTIF(`%%s` IS NULL) AS `%%s`', c.column_name, c.column_name), ', ') null_struct,
    STRING_AGG(FORMAT('COUNTIF(`%%s` IS NULL)', c.column_name), ' + ') null_sum
  FROM %s.COLUMNS c
  JOIN %s.TABLES tb USING (table_catalog, table_schema, table_name)
  WHERE tb.table_type = 'BASE TABLE' AND c.table_schema != 'dq'
    AND (@ds IS NULL OR c.table_schema = @ds)
  GROUP BY 1,2,3""", isq, isq)
USING ds AS ds;

-- 3. PROFILE EACH TABLE
FOR rec IN (SELECT * FROM dq._work) DO
  BEGIN
    EXECUTE IMMEDIATE FORMAT("""
      INSERT dq.table_profile
      SELECT '%s.%s.%s', COUNT(*),
             COUNT(*) - COUNT(DISTINCT TO_JSON_STRING(_src)),
             %s, TO_JSON_STRING(STRUCT(%s)), NULL, CURRENT_TIMESTAMP()
      FROM `%s.%s.%s` _src""",
      rec.p, rec.d, rec.t, rec.null_sum, rec.null_struct, rec.p, rec.d, rec.t);
  EXCEPTION WHEN ERROR THEN
    INSERT dq.table_profile (table_id, error, run_ts)
    VALUES (FORMAT('%s.%s.%s', rec.p, rec.d, rec.t), @@error.message, CURRENT_TIMESTAMP());
  END;
END FOR;

-- 4. COLLATED RESULT
EXECUTE IMMEDIATE FORMAT("""
CREATE OR REPLACE TABLE dq.data_completeness AS
WITH meta AS (
  SELECT CONCAT(project_id,'.',table_schema,'.',table_name) table_id,
         creation_time, storage_last_modified_time
  FROM %s.TABLE_STORAGE WHERE deleted = FALSE
),
cols AS (
  SELECT CONCAT(table_catalog,'.',table_schema,'.',table_name) table_id, COUNT(*) col_count
  FROM %s.COLUMNS GROUP BY 1
),
parts AS (
  SELECT CONCAT(table_catalog,'.',table_schema,'.',table_name) table_id,
         MIN(partition_id) first_partition, MAX(partition_id) last_partition,
         COUNT(*) partition_count
  FROM %s.PARTITIONS
  WHERE partition_id NOT IN ('__NULL__','__UNPARTITIONED__')
  GROUP BY 1
),
loads AS (
  SELECT CONCAT(destination_table.project_id,'.',destination_table.dataset_id,'.',destination_table.table_id) table_id,
         COUNT(*) loads_30d, MAX(creation_time) last_load,
         SAFE_DIVIDE(TIMESTAMP_DIFF(MAX(creation_time), MIN(creation_time), HOUR), COUNT(*) - 1) avg_hrs
  FROM %s.JOBS
  WHERE creation_time >= TIMESTAMP_SUB(CURRENT_TIMESTAMP(), INTERVAL 30 DAY)
    AND state = 'DONE' AND error_result IS NULL
    AND destination_table.table_id IS NOT NULL
    AND (job_type = 'LOAD' OR statement_type IN ('INSERT','MERGE','UPDATE','CREATE_TABLE_AS_SELECT'))
  GROUP BY 1
)
SELECT
  SPLIT(m.table_id,'.')[OFFSET(1)] AS dataset,
  m.table_id,
  m.creation_time AS history_start,
  m.storage_last_modified_time AS last_updated,
  DATE_DIFF(CURRENT_DATE(), DATE(m.storage_last_modified_time), DAY) AS days_since_update,
  p.first_partition, p.last_partition, p.partition_count,
  l.loads_30d, l.last_load, ROUND(l.avg_hrs, 1) AS avg_hrs_between_loads,
  CASE WHEN l.avg_hrs IS NULL THEN 'Unknown'
       WHEN l.avg_hrs <= 1.5 THEN 'Hourly'
       WHEN l.avg_hrs <= 30  THEN 'Daily'
       WHEN l.avg_hrs <= 200 THEN 'Weekly'
       ELSE 'Monthly+' END AS ingestion_frequency,
  pr.total_rows, c.col_count,
  pr.dup_rows, pr.dup_rows > 0 AS has_duplicates,
  pr.total_nulls,
  ROUND(100 * (1 - SAFE_DIVIDE(pr.total_nulls, pr.total_rows * c.col_count)), 2) AS completeness_pct,
  pr.null_by_column, pr.error
FROM meta m
LEFT JOIN cols c USING (table_id)
LEFT JOIN parts p USING (table_id)
LEFT JOIN loads l USING (table_id)
LEFT JOIN dq.table_profile pr USING (table_id)
WHERE STARTS_WITH(m.table_id, @scope)
  AND SPLIT(m.table_id,'.')[OFFSET(1)] != 'dq'
""", isq, isq, isq, isq, isq)
USING scope AS scope;

SELECT * FROM dq.data_completeness ORDER BY dataset, completeness_pct;
